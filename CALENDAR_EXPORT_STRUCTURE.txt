USAGE PATTERN
-------------

1. Import library:
   import { generateICalendar, MarketEvent } from '@/lib/calendar';

2. Build market events:
   const markets: MarketEvent[] = [
     { pool_id: 1, name: '...', end_time: ..., created_at: ..., ... }
   ];

3. Generate iCalendar:
   const ical = generateICalendar(markets, {
     title: 'My Markets',
     sourceBaseUrl: 'https://predifi.market'
   });

4. Export (browser):
   const blob = new Blob([ical], { type: 'text/calendar;charset=utf-8' });
   const url = URL.createObjectURL(blob);
   // Trigger download...

5. Validate (testing):
   const parsed = parseICalendar(ical);
   console.assert(parsed.length === markets.length);
